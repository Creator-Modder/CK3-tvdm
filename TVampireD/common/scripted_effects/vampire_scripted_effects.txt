add_vampire_lifestyle_xp_tick = {
	if = {
		limit = {
			has_trait = vampire
			has_trait_xp = {
				trait = vampire
				value < 100
			}
		}
		add_trait_xp = {
			trait = vampire
			value = 1
		}
	}
}

# Effect for turning someone into a vampire (includes healing)
# Now requires $SIRE$ parameter to establish bloodline
turn_into_vampire = {
    # Store the sire reference
    if = {
        limit = { always = yes }  # This parameter should always be provided
        set_variable = {
            name = sire_character
            value = $SIRE$
        }
		add_character_flag = has_sire
    }

    # Add vampire trait
    add_trait = vampire
	if = {
		limit = { 
			OR = {
				has_trait = compelled
				has_trait = original_compelled
			}
		}
		remove_trait = compelled
		remove_trait = original_compelled
	}
    # Initialize vampire XP tracking variable
    set_variable = {
        name = vampire_years_since_xp_gain
        value = 0
    }

    # Add highest level beauty trait
    if = {
        limit = { 
            NOR = {
                has_trait = beauty_good_1
                has_trait = beauty_good_2
                has_trait = beauty_good_3
            }
        }
        add_trait_force_tooltip = beauty_good_3  # Beautiful (highest level)
    }

    # Initialize vampire lifetime reset tracking
    vampire_init_years_since_lifetime_reset = yes

    # First, heal them completely
    vampire_regenerative_healing = yes
    vampire_initialize_visual_age = yes
}

# Set vampire visual age and lock it
vampire_set_visual_age = {
	set_variable = {
		name = vampire_visual_age
		value = $VISUALAGE$
	}
	set_immortal_age = var:vampire_visual_age
}

# Initialize vampire visual age when first turned
vampire_initialize_visual_age = {
	if = {
		limit = {
			NOT = { exists = var:vampire_visual_age }
		}
		# Set visual age to current age when turned (minimum 18, maximum 35)
		if = {
			limit = { age < 18 }
			vampire_set_visual_age = { VISUALAGE = 18 }
		}
		else_if = {
			limit = { age > 35 }
			vampire_set_visual_age = { VISUALAGE = 35 }
		}
		else = {
			vampire_set_visual_age = { VISUALAGE = age }
		}
	}
}

# Refresh vampire visual age (to prevent aging)
vampire_refresh_visual_age = {
	if = {
		limit = {
			exists = var:vampire_visual_age
		}
		set_immortal_age = var:vampire_visual_age
	}
	else = {
		# Fallback if variable doesn't exist
		vampire_initialize_visual_age = yes
	}
}


remove_compelled_when_modifier_expires = {
    if = {
        limit = { 
            has_trait = compelled
            NOT = { has_character_modifier = remove_compelled_trait }
        }
        remove_trait = compelled
    }
    if = {
        limit = { 
            has_trait = original_compelled
            NOT = { has_character_modifier = remove_original_compelled_trait }
        }
        remove_trait = original_compelled
    }
}

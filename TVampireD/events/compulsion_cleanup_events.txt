namespace = compulsion_cleanup

# Yearly check to remove expired compulsion traits(Redundant)
compulsion_cleanup.0001 = {
    type = character_event
    hidden = yes
    
    trigger = {
        OR = {
            AND = {
                has_trait = compelled
                NOT = { has_character_modifier = remove_compelled_trait }
            }
            AND = {
                has_trait = original_compelled
                NOT = { has_character_modifier = remove_original_compelled_trait }
            }
        }
    }
    
    immediate = {
        # Remove compelled trait if modifier expired
        if = {
            limit = { 
                has_trait = compelled
                NOT = { has_character_modifier = remove_compelled_trait }
            }
            remove_trait = compelled
        }
        
        # Remove original_compelled trait if modifier expired  
        if = {
            limit = { 
                has_trait = original_compelled
                NOT = { has_character_modifier = remove_original_compelled_trait }
            }
            remove_trait = original_compelled
        }
    }
}

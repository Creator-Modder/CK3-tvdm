# Vampire Muscle Preservation System (MALE ONLY)
vampire_muscularity = {
	usage = game
	priority = 15  # Higher priority than normal body changes

	vampire_preserved_muscle = {
		ignore_outfit_tags = yes
		dna_modifiers = {
			morph = {
				mode = modify
				gene = gene_bs_body_shape
				value = {
					value = 0

					# Enhanced muscularity for MALE vampires only
					if = {
						limit = {
							is_female = no  # Only apply to males
							OR = {
								has_trait = vampire
								has_trait = original_vampire
								has_trait = original_hybrid
							}
						}
						
						# Base vampire muscle definition
						add = 0.5  # Strong, athletic build
						

						# Account for any physical traits they had
						if = { limit = { has_trait = physique_good_3 } add = 0.2 }
						else_if = { limit = { has_trait = physique_good_2 } add = 0.15 }
						else_if = { limit = { has_trait = physique_good_1 } add = 0.1 }
						
						if = { limit = { has_trait = strong } add = 0.1 }
						if = { limit = { has_trait = athletic } add = 0.05 }
						
						# Override any negative physical traits
						if = { limit = { has_trait = physique_bad_1 } add = 0.4 }  # Cancel and enhance
						if = { limit = { has_trait = physique_bad_2 } add = 0.6 }  # Cancel and enhance
						if = { limit = { has_trait = physique_bad_3 } add = 0.8 }  # Cancel and enhance
						if = { limit = { has_trait = weak } add = 0.4 }
					}

					max = 1.0
					min = 0.0  # Allow normal range for females, enhanced minimum for males
				}
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 1000  # Guaranteed application
				is_female = no  # Only apply to males
				OR = {
					has_trait = vampire
					has_trait = original_vampire
					has_trait = original_hybrid
				}
			}
		}
	}
}

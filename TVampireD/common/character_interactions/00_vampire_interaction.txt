vampire_compulsion_interaction = {
    category = interaction_category_friendly
	

    icon = "vampire_compulsion_interaction"
	
	desc = vampire_compulsion_interaction_desc
	
	interface_priority = 50
	common_interaction = yes
	
	is_shown = {
		scope:actor = { has_trait = vampire }
		scope:recipient = {
			NOR = {
				has_trait = vampire
				has_trait = original_vampire
                has_trait = original_hybrid
                #Other traits for the future (werewolves, witches, etc.)
			}
			is_alive = yes
            OR = {
                is_vassal_of = scope:actor
                is_courtier_of = scope:actor
                is_imprisoned_by = scope:actor
                is_spouse_of = scope:actor
                is_concubine_of = scope:actor
                is_parent_of = scope:actor
                is_child_of = scope:actor
                is_sibling_of = scope:actor
                is_knight_of = scope:actor
                is_pool_guest_of = scope:actor
                is_at_same_location = scope:actor
                scope:actor = { is_pool_guest_of = scope:recipient }
                AND = {
                    is_pool_guest = yes
                    scope:actor = { is_pool_guest = yes }
                    host = { 
                        any_pool_guest = { this = scope:actor }
                        any_pool_guest = { this = scope:recipient }
                    }
                }
            }
		}
	}
	
	is_valid = {
		scope:recipient = { 
			NOR = {
			    has_trait = compelled  # Can't compel someone already compelled
			    has_trait = original_compelled  # Can't compel someone already compelled by original
		    }
		}
	}
	
	on_accept = {
        scope:actor = {
            # Add loyalty hook with proper syntax
			add_hook = {
				type = loyalty_hook
				target = scope:recipient
			}
        }   
		scope:recipient = {
			# Add compelled trait
			add_trait = compelled
			
            # Add a modifier that will remove the trait
            hidden_effect = {
                add_character_modifier = {
                    modifier = remove_compelled_trait
                    years = 40
                }
            }

			# Add named opinion modifier
			add_opinion = {
				target = scope:actor             
				modifier = compelled_opinion
			}
			
			# Add stress loss (they're compelled, so they feel "better")
			add_stress = minor_stress_impact_loss
		}
	}
	
	# This makes it automatically succeed
	auto_accept = {
		always = yes
	}
	
	# AI will use this interaction
    ai_frequency = 12
    ai_potential = { has_trait = vampire }
    ai_targets = {
        ai_recipients = spouses
        ai_recipients = children
        ai_recipients = family
        ai_recipients = dynasty
        max = 6
    }
    ai_targets = {
        ai_recipients = neighboring_rulers
        ai_recipients = peer_vassals
        ai_recipients = liege
        max = 3
    }
    ai_targets = {
        ai_recipients = guests
        max = 9
    }
    ai_targets = {
        ai_recipients = councillors
        max = 9
    }
    ai_targets = {
        ai_recipients = prisoners
        max = 9
    }
    ai_targets = {
        ai_recipients = realm_characters
        max = 9
    }
    ai_targets = {
        ai_recipients = activity_guests
        max = 9
    }
    ai_targets = {
        ai_recipients = courtiers
        max = 9
    }
    ai_targets = {
        ai_recipients = vassals
        max = 5
    }

	ai_target_quick_trigger = {
		adult = yes
	}     
	ai_will_do = {
		base = 95
        modifier = {
            add = 25
            scope:recipient = {
                opinion = {
                    target = scope:actor
                    value < 1
                }
            }
        }
        modifier = {
            add = -1000
            scope:recipient = {
                is_ai = no
            }
        }
	}
}

# Turn Vampire Interactions

turn_vampire_interaction = {
    category = interaction_category_hostile
    

    icon = "turn_vampire_interaction"

    desc = turn_vampire_interaction_desc

    interface_priority = 70
    common_interaction = yes

    is_shown = {
        scope:actor = {
            OR = {
                has_trait = vampire
                has_trait = original_vampire
                has_trait = original_hybrid
            }
            is_alive = yes
            is_imprisoned = no
        }
        scope:recipient = {
            NOR = { 
                has_trait = original_vampire
                has_trait = vampire
                has_trait = original_hybrid
                # other supernatural beings (e.g., werewolves) that can't be turned will be added later
            }
            is_alive = yes
            is_adult = yes
            age >= 18
            is_pregnant = no
            OR = {
                is_vassal_of = scope:actor
                is_courtier_of = scope:actor
                is_imprisoned_by = scope:actor
                is_spouse_of = scope:actor
                is_concubine_of = scope:actor
                is_parent_of = scope:actor
                is_child_of = scope:actor
                is_sibling_of = scope:actor
                is_knight_of = scope:actor
                is_pool_guest_of = scope:actor
                is_at_same_location = scope:actor
                scope:actor = { is_pool_guest_of = scope:recipient }
                AND = {
                    is_pool_guest = yes
                    scope:actor = { is_pool_guest = yes }
                    host = { 
                        any_pool_guest = { this = scope:actor }
                        any_pool_guest = { this = scope:recipient }
                    }
                }
            }
        }
    }
    on_accept = {
        scope:actor = {
            # Remove loyalty hook with proper syntax
            remove_hook = {
                type = loyalty_hook
                target = scope:recipient
            }
        }
        scope:recipient = {
            # Add vampire trait with sire reference
            turn_into_vampire = { SIRE = scope:actor }

            # Remove compelled trait if present
            remove_trait = compelled
            remove_trait = original_compelled

            # Add stress loss (they're turned, so they feel "better")
            add_stress = minor_stress_impact_gain
        }
    }
    # This makes it automatically succeed
    auto_accept = {
        always = yes
    }
    # AI will use this interaction
    ai_frequency = 36

    ai_potential = { 
        has_trait = vampire
        has_trait = original_vampire
        has_trait = original_hybrid
    }
    ai_targets = {
        ai_recipients = spouses
        ai_recipients = children
        ai_recipients = family
        ai_recipients = dynasty
        max = 6
    }
    ai_targets = {
        ai_recipients = courtiers
        max = 4
    }
    ai_targets = {
        ai_recipients = realm_characters
        max = 9
    }
    ai_targets = {
        ai_recipients = vassals
        max = 5
    }

	ai_target_quick_trigger = {
		adult = yes
	}
    ai_will_do = {
        base = 100
        # More likely to use on characters that are friendly or have a high opinion of the vampire
        modifier = {
            add = 55
            scope:recipient = {
                opinion = {
                    target = scope:actor
                    value > 80
                }
                OR = {
                    has_relation_friend = scope:actor
                    has_relation_lover = scope:actor
                }
            }
        }
        modifier = {
            add = -1000
            scope:recipient = {
                is_ai = no
            }
        }
    }
}

    }
    
    is_valid_showing_failures_only = {
        scope:actor = {
            is_alive = yes
        }
        scope:recipient = {
            is_alive = yes
            # Check if they're within interaction range
            OR = {
                # Direct relationships
                is_vassal_of = scope:actor
                is_courtier_of = scope:actor

                is_parent_of = scope:actor
                is_child_of = scope:actor

                is_at_same_location = scope:actor
                is_pool_guest_of = scope:actor

                AND = {
                    is_pool_guest = yes

                        any_pool_guest = { this = scope:actor }

                    }
                }
                
                # Liege relationship
                scope:actor = { is_vassal_of = scope:recipient }
                
                # Peer vassals (same liege)

                    exists = liege
                    scope:actor = { exists = liege }

                    is_ruler = yes
                    scope:actor = { is_ruler = yes }
                    any_neighboring_realm_same_rank_owner = { this = scope:actor }
                }
            }
        }

                    has_trait = consumption
                    has_trait = incapable
                    has_trait = great_pox
                    has_trait = pneumonic
                    has_trait = measles
                    has_trait = dysentery
                    has_trait = typhus
                }
            }
            desc = "Desperately needs healing"
        }
        
        # More likely if they like the vampire
        opinion_modifier = {
            opinion_target = scope:actor
            multiplier = 1.5
        }
        
        # Less likely if they're suspicious
        modifier = {
            add = -30
            scope:recipient = {
                OR = {
                    has_trait = paranoid
                    has_trait = zealous
                }
            }
            desc = "Suspicious of supernatural aid"
        }
        
        # More likely if they're friends/family
        modifier = {
            add = 40
            scope:recipient = {

                    is_close_family_of = scope:actor

request_vampire_turn_interaction = {
    category = interaction_category_friendly
    icon = "request_vampire_turn_interaction"
    
    desc = request_vampire_turn_interaction_desc
    
    interface_priority = 60
    common_interaction = yes
    
    is_shown = {
        scope:actor = {
            # Actor (requester) must be human
            NOR = {
                has_trait = vampire
                has_trait = original_vampire
                has_trait = original_hybrid
            }
            is_adult = yes
            age >= 18
            is_pregnant = no
            is_imprisoned = no
        }
        scope:recipient = {
            # Recipient must be a vampire
            OR = {
                has_trait = vampire
                has_trait = original_vampire
                has_trait = original_hybrid
            }
            is_alive = yes
            # Must have some relationship/proximity
            OR = {
                is_vassal_of = scope:actor
                is_courtier_of = scope:actor
                scope:actor = { is_vassal_of = scope:recipient }
                scope:actor = { is_courtier_of = scope:recipient }
                is_spouse_of = scope:actor
                is_concubine_of = scope:actor
                is_parent_of = scope:actor
                is_child_of = scope:actor
                is_sibling_of = scope:actor
                is_knight_of = scope:actor
                scope:actor = { is_knight_of = scope:recipient }
                is_pool_guest_of = scope:actor

                is_at_same_location = scope:actor
                AND = {
                    is_pool_guest = yes
                    scope:actor = { is_pool_guest = yes }
                    host = { 
                        any_pool_guest = { this = scope:actor }
                        any_pool_guest = { this = scope:recipient }
                    }
                }
            }
        }
    }

    is_valid_showing_failures_only = {
        scope:actor = {
            is_alive = yes
        }
        scope:recipient = {
            is_alive = yes
        }
    }
    
    # Send options - these create the multiple choice interface
    send_option = {
        is_shown = {
            scope:recipient = {
                opinion = {
                    target = scope:actor
                    value >= 80
                }
            }
        }
        flag = high_opinion
        localization = REQUEST_TURN_HIGH_OPINION
    }
    
    send_option = {
        is_valid = {
            scope:actor = {
                has_strong_hook = scope:recipient
            }
        }
        flag = hook
        localization = REQUEST_TURN_STRONG_HOOK
    }
    
    send_option = {
        is_shown = {
            scope:actor = {
                gold >= 1000
            }
        }
        flag = gold_payment
        localization = REQUEST_TURN_GOLD_PAYMENT
    }

    send_option = {
        is_valid = {
            scope:actor = {
                has_weak_hook = scope:recipient
            }
        }
        flag = weak_hook
        localization = REQUEST_TURN_WEAK_HOOK
    }
    
    
    should_use_extra_icon = {
        scope:actor = { has_usable_hook = scope:recipient }
    }
    extra_icon = "hook_icon"
    
    send_options_exclusive = no
    
    # Cost system
    cost = {
        gold = {
            value = 0
            if = {
                limit = { scope:gold_payment = yes }
                add = 1000
                desc = GOLD_PAYMENT_COST
            }
        }
    }

            limit = { scope:hook = yes }

            }

            limit = { scope:weak_hook = yes }

                target = scope:recipient
                modifier = granted_vampirism_opinion
            }

            trigger_event = {
                id = vampire_turn_request.0002
                days = 1
            }
        }
        
        # Sire gains opinion of their new childe and stress
        scope:recipient = {
            add_opinion = {
                target = scope:actor
                modifier = new_childe_opinion
            }
        }
        
        # Send notification message to sire
        scope:recipient = {
            send_interface_message = {
                type = event_generic_neutral
                title = vampire_turn_request_accept_notification
                right_icon = scope:actor

		prestige >= 2000

				}
			}
			any_close_family_member = {
				has_trait = witch
			}
		}

	}

		step_two_flag = "BECOME_ORIGINAL_VAMPIRE_DECISION_CONFIRM"
		
		is_visible_step_two = {
			gold >= 1000
			prestige >= 2000
			piety >= 1500
			OR = {
				any_courtier_or_guest = {
					has_relation_best_friend  = root
					has_trait = witch
				}
				any_close_family_member = {
					has_trait = witch
				}
			}
		}

		step_two_button_positive = "BECOME_ORIGINAL_VAMPIRE_DECISION_CONFIRM_ACCEPT"
		step_two_button_negative = "GENERIC_CANCEL"
	}

	effect = {
		trigger_event = {
			id = vampire_transformation.0001
		}
		show_as_tooltip = {
			add_trait = original_vampire
		}	
	}
	
	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}

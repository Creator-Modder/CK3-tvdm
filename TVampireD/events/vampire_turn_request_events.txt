namespace = vampire_turn_request

# Event for successful vampire transformation
vampire_turn_request.0002 = {
    type = character_event
    title = vampire_turn_request.0002.t
    desc = vampire_turn_request.0002.desc

    theme = intrigue
    
    left_portrait = {
        character = root              # New vampire (the requester)
        animation = personality_bold
    }

    option = {
        name = vampire_turn_request.0002.a
        
        # Embrace the new vampiric nature
        add_stress = minor_stress_impact_loss
        
        # Chance to gain ambitious trait if they don't have personality conflicts
        if = {
            limit = { 
                NOR = { 
                    has_trait = ambitious
                    has_trait = content
                } 
            }
            random = {
                chance = 10
                add_trait = ambitious
            }
        }
        
        ai_chance = {
            base = 60
            
            modifier = {
                add = 30
                has_trait = ambitious
            }
            
            modifier = {
                add = -40
                has_trait = content
            }
        }
    }

    option = {
        name = vampire_turn_request.0002.b
        
        # Feel conflicted about the transformation
        add_stress = minor_stress_impact_gain
        
        # Small chance to develop depression from the change
        if = {
            limit = { 
                NOT = { has_trait = depressed_1 }
            }
            random = {
                chance = 10
                add_trait = depressed_1
            }
        }
        
        ai_chance = {
            base = 40
            
            modifier = {
                add = 20
                has_trait = content
            }
            
            modifier = {
                add = 15
                has_trait = compassionate
            }
            
            modifier = {
                add = -25
                has_trait = callous
            }
        }
    }
    option = {
        name = vampire_turn_request.0002.c

        ai_chance = {
            base = 5
            
            modifier = {
                add = 20
                has_trait = diligent
            }
            
            modifier = {
                add = 15
                has_trait = arbitrary
            }
        }
    }
}
